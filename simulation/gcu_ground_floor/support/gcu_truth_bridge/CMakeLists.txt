cmake_minimum_required(VERSION 3.16)
project(gcu_truth_bridge)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(ros_gz_bridge REQUIRED)
find_package(ignition-transport11 REQUIRED)
find_package(ignition-msgs8 REQUIRED)
add_executable(gcu_truth_bridge truth_bridge.cpp)
ament_target_dependencies(gcu_truth_bridge rclcpp tf2_msgs)
target_link_libraries(gcu_truth_bridge ignition-transport11::core ignition-msgs8::ignition-msgs8)
install(TARGETS gcu_truth_bridge DESTINATION lib/${PROJECT_NAME})
add_executable(gcu_ordered_bridge ordered_bridge.cpp)
ament_target_dependencies(gcu_ordered_bridge rclcpp ros_gz_bridge)
install(TARGETS gcu_ordered_bridge DESTINATION lib/${PROJECT_NAME})
ament_package()
