cmake_minimum_required(VERSION 3.16)
project(gcu_tf_probe)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(gcu_tf_probe tf_probe.cpp)
ament_target_dependencies(gcu_tf_probe rclcpp tf2_msgs std_msgs)
install(TARGETS gcu_tf_probe DESTINATION lib/${PROJECT_NAME})
ament_package()
